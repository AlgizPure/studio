## 🚀 Как использовать новые паттерны

### Для новых компонентов:

```tsx
// 1. Используй useFirestoreMutation для всех мутаций
import { useFirestoreCreate, useFirestoreUpdate, useFirestoreDelete } from '@/firebase/firestore/use-firestore-mutation';

const { mutate: create } = useFirestoreCreate({
  successMessage: 'Создано',
  errorMessage: 'Ошибка',
});

// 2. Оберни страницу в AsyncBoundary
import { AsyncBoundary } from '@/components/ui/async-boundary';

return (
  <AsyncBoundary>
    <YourComponent />
  </AsyncBoundary>
);

// 3. Для больших списков используй пагинацию
import { usePaginatedCollection } from '@/firebase/firestore/use-paginated-collection';

const { data, loadMore, hasMore } = usePaginatedCollection(query, {
  pageSize: 20,
});
```
